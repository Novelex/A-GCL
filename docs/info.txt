python
import numpy as np, scipy.io as sio, os
from sklearn.model_selection import StratifiedKFold, GridSearchCV
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import StandardScaler
from sklearn.svm import LinearSVC
from sklearn.metrics import accuracy_score, roc_auc_score

X, y = [], []
for label, d in [(1, 'data/raw/ASD_ADJ'), (0, 'data/raw/NC_ADJ')]:
    for f in sorted(os.listdir(d)):
        fc = np.nan_to_num(sio.loadmat(os.path.join(d, f))['cropped_matrix'])
        X.append(fc[np.triu_indices_from(fc, k=1)]); y.append(label)
X, y = np.array(X), np.array(y)
print("X:", X.shape, "ASD:", y.sum(), "NC:", len(y)-y.sum())
accs, aucs = [], []
for tr, te in StratifiedKFold(5, shuffle=True, random_state=123).split(X, y):
    clf = GridSearchCV(Pipeline([('s', StandardScaler()), ('c', LinearSVC(dual=False, max_iter=10000))]),
                       {'c__C': [0.001, 0.01, 0.1, 1, 10, 100, 1000]}, cv=5, n_jobs=16)
    clf.fit(X[tr], y[tr])
    accs.append(accuracy_score(y[te], clf.predict(X[te])))
    aucs.append(roc_auc_score(y[te], clf.decision_function(X[te])))
print("raw-FC SVM  acc %.4f ± %.4f   auc %.4f ± %.4f" % (np.mean(accs), np.std(accs), np.mean(aucs), np.std(aucs)))
